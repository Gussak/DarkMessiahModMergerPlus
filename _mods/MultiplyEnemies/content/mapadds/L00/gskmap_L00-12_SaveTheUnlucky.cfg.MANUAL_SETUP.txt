"mapadd_keyvalues"
{
	"PreInitialize"
	{	

		// 1. Create the survival tracker switch
		add:entity
		{
			"classname" "logic_branch"
			"targetname" "SaveTheUnlucky_friendly_survival_check"
			"InitialValue" "1" // Starts as True (Survived)
		}
		modify:entity
		{
			"TargetMarkers"
			{
				"targetname"	"SaveTheUnlucky_friendly_survival_check"
			}
			"add:key"
			{
				// If the check passes when tested, award the skill point
				"OnTrue" "gsk_generic_cmds_issuer,Command,mm_player_add_skillpoints 1,0,-1"
				"OnTrue" "gsk_generic_cmds_issuer,Command,echo The Unlucky Shrinked Guard was saved -- you get 1 skill point,0,-1"
				"OnFalse" "gsk_generic_cmds_issuer,Command,echo The Unlucky Shrinked Guard DIED -- you get NO skill point,0,-1"
			}
		}
		// 2. Create the counter entity (Modified to trigger the test)
		add:entity
		{
			"classname" "math_counter"
			"targetname" "SaveTheUnlucky_kill_counter"
			"startvalue" "0"
			"max" "31" //the total found at related MapAdds auto generated file SYNC_HELPER_TIP
		}
		modify:entity
		{
			"TargetMarkers"
			{
				"targetname"	"SaveTheUnlucky_kill_counter"
			}
			"add:key"
			{
				// When all foes are dead, don't give the point automatically—test the branch instead
				"OnHitMax" "SaveTheUnlucky_friendly_survival_check,Test,,0,-1"
			}
		}
		//// 3. Create a server command executor
		//add:entity
		//{
			//"classname" "point_servercommand"
			//"targetname" "server_command" //replaced by gsk_generic_cmds_issuer
		//}
		// 4. Hook the friendly NPC to the survival tracker
		modify:entity
		{
			"TargetMarkers"
			{
				"targetname" "gskSpawn_12_SaveTheUnlucky_000" //the sole mini guard auto generated name, just keep it as first one there SYNC_HELPER_TIP
			}
			"add:key"
			{
				// If this specific friendly dies, flip the switch to False (0)
				"OnDeath" "SaveTheUnlucky_friendly_survival_check,SetValue,0,0,-1"
			}
		}
		// 5. Hook your ambush enemies to the kill counter (Example)
		modify:entity
		{
			"TargetMarkers"
			{
				//"targetname" "gskSpawn_12_SaveTheUnlucky_001" //the mini spiders
				"classname" "npc_spider_mini" //it can be set to other minispiders elsewhere, but if player kills enough it will work
			}
			"add:key"
			{
				// If this specific friendly dies, flip the switch to False (0)
				"OnDeath" "SaveTheUnlucky_kill_counter,Add,1,0,-1"
			}
		}

	}
}
