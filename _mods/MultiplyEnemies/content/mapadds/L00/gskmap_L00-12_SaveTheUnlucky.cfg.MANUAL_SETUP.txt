"mapadd_keyvalues"
{
	"PreInitialize"
	{	

			// ==========================================
			// 1. BASE ENTITY INITIALIZATION (CREATION)
			// ==========================================

			// Create the survival tracker switch
			add:entity
			{
					"classname" "logic_branch"
					"targetname" "SaveTheUnlucky_friendly_survival_check"
					"InitialValue" "1"
			}

			// Create the counter entity
			add:entity
			{
					"classname" "math_counter"
					"targetname" "SaveTheUnlucky_kill_counter"
					"startvalue" "0"
					"max" "31"
			}

			// ==========================================
			// 2. LOGIC HOOKS & OUTPUT LINKS (MODIFICATIONS)
			// ==========================================

			// Hook the logic outputs onto the survival tracker switch
			modify:entity
			{
					TargetMarkers
					{
							"classname" "logic_branch"
							"targetname" "SaveTheUnlucky_friendly_survival_check"
					}
					add:key
					{
							"OnTrue" "gsk_generic_cmds_issuer,Command,mm_player_add_skillpoints 1,0,-1"
							"OnFalse" "gsk_generic_cmds_issuer,Command,echo \"NPC DIED - NO SKILL POINT\",0,-1"
					}
			}

			// Hook the conditional check to fire when the kill counter finishes
			modify:entity
			{
					TargetMarkers
					{
							"classname" "math_counter"
							"targetname" "SaveTheUnlucky_kill_counter"
					}
					add:key
					{
							"OnHitMax" "SaveTheUnlucky_friendly_survival_check,Test,,0,-1"
					}
			}

			// Hook the friendly NPC to the survival tracker switch
			modify:entity
			{
					TargetMarkers
					{
							"targetname" "gskSpawn_12_SaveTheUnlucky_000"
					}
					add:key
					{
							"OnDeath" "SaveTheUnlucky_friendly_survival_check,SetValue,0,0,-1"
					}
			}

			// Hook all mini spiders globally to the kill counter
			modify:entity
			{
					TargetMarkers
					{
							"classname" "npc_spider_mini"
					}
					add:key
					{
							"OnDeath" "SaveTheUnlucky_kill_counter,Add,1,0,-1"
					}
			}

	}
}
